  <script>
    const SCRIPT_URL = 'วางลิงก์_WEB_APP_URL_ตรงนี้';
    let allOrders = [];

    async function loadOrders() {
      const tbody = document.getElementById('orderTable');
      tbody.innerHTML = '<tr><td colspan="6" class="p-8 text-center text-slate-500">กำลังดึงข้อมูลจาก Google Sheets...</td></tr>';
      
      try {
        const res = await fetch(SCRIPT_URL);
        allOrders = await res.json();
        
        if (allOrders.length === 0) {
          tbody.innerHTML = '<tr><td colspan="6" class="p-8 text-center text-slate-500">ยังไม่มีรายการสั่งซื้อ</td></tr>';
          return;
        }

        tbody.innerHTML = '';
        allOrders.slice().reverse().forEach((ord) => {
          const isPacked = String(ord.Status).includes('Packed');
          tbody.innerHTML += `
            <tr class="hover:bg-slate-50">
              <td class="p-4"><span class="font-bold text-teal-700">${ord.OrderID}</span><br><span class="text-xs text-slate-400">${ord.Timestamp}</span></td>
              <td class="p-4"><p class="font-bold">${ord.CustomerName}</p><p class="text-xs text-slate-600">โทร: ${ord.Phone}</p><p class="text-xs text-slate-500 mt-1 max-w-xs">${ord.Address}</p></td>
              <td class="p-4 space-y-1">
                ${ord.SKU1_Qty > 0 ? `<div class="bg-teal-50 border border-teal-200 text-teal-900 px-2 py-1 rounded text-xs">✅ หยิบ <b>SKU-01 (Soft)</b>: <b>${ord.SKU1_Qty}</b> ชิ้น</div>` : ''}
                ${ord.SKU2_Qty > 0 ? `<div class="bg-slate-100 border border-slate-300 text-slate-900 px-2 py-1 rounded text-xs">✅ หยิบ <b>SKU-02 (Charcoal)</b>: <b>${ord.SKU2_Qty}</b> ชิ้น</div>` : ''}
              </td>
              <td class="p-4 font-bold">${ord.TotalPrice} ฿</td>
              <td class="p-4"><span class="px-2.5 py-1 rounded-full text-xs font-bold ${isPacked ? 'bg-green-100 text-green-700' : 'bg-amber-100 text-amber-700'}">${ord.Status}</span></td>
              <td class="p-4 text-center space-y-2">
                <button onclick="printLabel('${ord.OrderID}')" class="w-full bg-slate-800 text-white px-3 py-1.5 rounded text-xs font-bold">🖨️ พิมพ์ใบจ่าหน้า</button>
                ${!isPacked ? `<button onclick="markPacked('${ord.OrderID}')" class="w-full bg-green-600 text-white px-3 py-1.5 rounded text-xs font-bold">📦 ยืนยันแพ็คแล้ว</button>` : '<p class="text-xs text-green-600 font-bold">แพ็คเรียบร้อย</p>'}
              </td>
            </tr>
          `;
        });
      } catch (err) {
        tbody.innerHTML = '<tr><td colspan="6" class="p-8 text-center text-red-500">โหลดข้อมูลไม่สำเร็จ กรุณาตรวจสอบ SCRIPT_URL</td></tr>';
      }
    }

    function printLabel(orderId) {
      const ord = allOrders.find(o => o.OrderID === orderId);
      if (!ord) return;

      document.getElementById('lblOrderId').innerText = ord.OrderID;
      document.getElementById('lblName').innerText = ord.CustomerName;
      document.getElementById('lblPhone').innerText = 'โทร: ' + ord.Phone;
      document.getElementById('lblAddress').innerText = 'ที่อยู่: ' + ord.Address;
      
      let items = [];
      if (ord.SKU1_Qty > 0) items.push(`SKU-01 Ultra Soft (${ord.SKU1_Qty} ชิ้น)`);
      if (ord.SKU2_Qty > 0) items.push(`SKU-02 Bamboo Charcoal (${ord.SKU2_Qty} ชิ้น)`);
      document.getElementById('lblItems').innerText = items.join(' + ');

      const printArea = document.getElementById('printArea');
      printArea.classList.remove('hidden');
      window.print();
      printArea.classList.add('hidden');
    }

    async function markPacked(orderId) {
      if (!confirm(`ยืนยันว่าหยิบและแพ็คสินค้าออเดอร์ ${orderId} เรียบร้อยแล้ว?`)) return;
      await fetch(SCRIPT_URL, { method: 'POST', body: JSON.stringify({ action: 'pack', orderId: orderId }) });
      loadOrders();
    }

    loadOrders();
  </script>
